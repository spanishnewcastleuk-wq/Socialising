(function(){
'use strict';
const activities=['ir al cine','tomar un café','ir a un concierto','visitar un museo','comer en un restaurante','pasear por el parque','ir de compras','jugar al tenis','estudiar juntos','ir a una fiesta'];
const days=['el lunes','el martes','el miércoles','el jueves','el viernes','el sábado','el domingo','mañana'];
const hours=['a las diez','a las doce','a las tres','a las cuatro','a las cinco','a las seis','a las siete','a las ocho'];
const responses=['Acepta con entusiasmo.','No puedes: tienes otros planes.','Propón otra hora.','Propón otro día.','No te apetece. Explica por qué.','Sugiere otro lugar.','Acepta, pero propone otra actividad.','Estás ocupado/a. Sugiere el domingo.','Acepta y pregunta dónde quedáis.','Rechaza la invitación educadamente.'];
const root=document.getElementById('lps1101-ruleta');
if(!root || root.dataset.ready==='true') return;
root.dataset.ready='true';
const el=id=>root.querySelector('[id=+id+]');
const choose=items=>items[Math.floor(Math.random()*items.length)];
let round=1,secret=choose(responses),shown=false,seconds=90,interval=null;
function clock(){el('clock').textContent=String(Math.floor(seconds/60)).padStart(2,'0')+':'+String(seconds%60).padStart(2,'0')}
function stop(){if(interval!==null){clearInterval(interval);interval=null}el('start').textContent='Iniciar'}
function reset(){stop();seconds=90;clock();el('status').textContent=''}
function reveal(){shown=!shown;el('response').textContent=shown?secret:'Respuesta oculta';el('reveal').textContent=shown?'Ocultar respuesta':'Mostrar respuesta (B)';el('reveal').setAttribute('aria-pressed',String(shown))}
function spin(){el('activity').textContent=choose(activities);el('day').textContent=choose(days);el('hour').textContent=choose(hours);secret=choose(responses);if(shown)reveal();round++;el('round').textContent='Ronda '+round;reset()}
function start(){if(interval!==null){stop();return}if(seconds===0)seconds=90;clock();el('start').textContent='Pausar';el('status').textContent='';interval=setInterval(()=>{seconds--;clock();if(seconds<=0){stop();el('start').textContent='Repetir';el('status').textContent='¡Tiempo! Cambiad de pareja o girad la ruleta.'}},1000)}
el('spin').addEventListener('click',spin);
el('reveal').addEventListener('click',reveal);
el('start').addEventListener('click',start);
el('reset').addEventListener('click',reset);
clock();
})();
</script>
